cmake_minimum_required(VERSION 3.16)
project(phyloSMC VERSION 0.1.0 LANGUAGES C)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
set(PHYLOSMC_CURUPIXA_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/external/curupixa" CACHE PATH "Curupixa checkout")
if(CMAKE_SOURCE_DIR STREQUAL PROJECT_SOURCE_DIR)
  set(phylosmc_top_level ON)
else()
  set(phylosmc_top_level OFF)
endif()
option(PHYLOSMC_BUILD_TESTS "Build regression tests" ${phylosmc_top_level})
option(PHYLOSMC_INSTALL "Generate installation rules" ${phylosmc_top_level})
if(NOT TARGET curupixa::inference)
  set(CURUPIXA_INSTALL ${PHYLOSMC_INSTALL} CACHE BOOL "Install Curupixa with phyloSMC")
  add_subdirectory("${PHYLOSMC_CURUPIXA_SOURCE}" external-build)
endif()
add_library(phylosmc src/phylosmc.c src/particle.c src/moves.c src/observations.c src/parameters.c src/rooting.c src/workers.c)
add_library(phyloSMC::phylosmc ALIAS phylosmc)
target_compile_features(phylosmc PUBLIC c_std_17)
target_include_directories(phylosmc PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include/phyloSMC>)
target_link_libraries(phylosmc PUBLIC curupixa::inference)
add_executable(phyloSMC src/main.c)
target_link_libraries(phyloSMC PRIVATE phylosmc)
if(PHYLOSMC_BUILD_TESTS)
  enable_testing()
  add_executable(test_extended tests/test_extended.c)
  target_compile_options(test_extended PRIVATE -UNDEBUG)
  target_link_libraries(test_extended PRIVATE phylosmc)
  add_test(NAME extended COMMAND test_extended)
  add_executable(test_smc tests/test_smc.c)
  target_compile_options(test_smc PRIVATE -UNDEBUG)
  target_link_libraries(test_smc PRIVATE phylosmc)
  add_test(NAME smc COMMAND test_smc)
  add_executable(example_api examples/api.c)
  target_link_libraries(example_api PRIVATE phylosmc)
  add_test(NAME api_example COMMAND example_api)
  add_test(NAME cli_anneal COMMAND phyloSMC ${CMAKE_CURRENT_SOURCE_DIR}/examples/tiny.fasta --particles 64 --moves 4)
  add_test(NAME cli_online COMMAND phyloSMC ${CMAKE_CURRENT_SOURCE_DIR}/examples/tiny.fasta --particles 64 --initial-taxa 2)
  add_test(NAME cli_fixed COMMAND phyloSMC ${CMAKE_CURRENT_SOURCE_DIR}/examples/tiny.fasta --particles 32 --root-mode fixed)
  add_test(NAME cli_outgroup COMMAND phyloSMC ${CMAKE_CURRENT_SOURCE_DIR}/examples/tiny.fasta --particles 32 --root-mode fixed --outgroup A,B)
  add_test(NAME cli_movable COMMAND phyloSMC ${CMAKE_CURRENT_SOURCE_DIR}/examples/tiny.fasta --particles 32 --root-mode movable)
  if(UNIX)
    add_test(NAME cli_processes COMMAND phyloSMC ${CMAKE_CURRENT_SOURCE_DIR}/examples/tiny.fasta --particles 32 --workers 2 --model gtr)
  endif()
  add_test(NAME cli_clock COMMAND phyloSMC ${CMAKE_CURRENT_SOURCE_DIR}/examples/tiny.fasta --particles 64 --clock lognormal)
endif()
if(PHYLOSMC_INSTALL)
  install(TARGETS phylosmc phyloSMC EXPORT phyloSMCTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
  install(FILES LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/phyloSMC)
  install(FILES README.md DESTINATION ${CMAKE_INSTALL_DOCDIR})
  install(DIRECTORY docs/ DESTINATION ${CMAKE_INSTALL_DOCDIR}/docs)
  install(FILES include/phylosmc.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/phyloSMC)

  install(EXPORT phyloSMCTargets NAMESPACE phyloSMC:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/phyloSMC)
  configure_package_config_file(cmake/phyloSMCConfig.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/phyloSMCConfig.cmake"
    INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/phyloSMC)
  write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/phyloSMCConfigVersion.cmake"
    VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
  install(FILES "${CMAKE_CURRENT_BINARY_DIR}/phyloSMCConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/phyloSMCConfigVersion.cmake"
    DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/phyloSMC)
endif()

if(APPLE)
  set(phylosmc_origin "@loader_path")
elseif(UNIX)
  set(phylosmc_origin "$ORIGIN")
endif()
if(DEFINED phylosmc_origin)
  file(RELATIVE_PATH phylosmc_relative_lib "${CMAKE_INSTALL_FULL_BINDIR}" "${CMAKE_INSTALL_FULL_LIBDIR}")
  set_target_properties(phylosmc PROPERTIES INSTALL_RPATH "${phylosmc_origin}")
  set_target_properties(phyloSMC PROPERTIES INSTALL_RPATH "${phylosmc_origin}/${phylosmc_relative_lib}")
endif()
